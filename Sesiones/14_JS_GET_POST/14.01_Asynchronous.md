## Método POST
1. En el archivo views.py crear el método get_cities 
```python
import json
...
@require_POST
def get_cities(request):
    data = json.loads(request.body)
    country = data.get('country', '')
    print(f'Pais: {country}')
    if country == 'México':
        cities = ['Ciudad de México', 'Guadalajara', 'Monterrey']
    elif country == 'USA':
        cities = ['Nueva York', 'Los Ángeles', 'Çhicago']
    elif country == 'Canada':
        cities = ['Vacouver', 'Montreal', 'Quebec']
    else:
        cities = []
    return JsonResponse({'cities': cities})
```

function getCookie(name) {
    let cookieArr = document.cookie.split(';');
    for (let i=0; i<cookieArr.length; i++) {
      let cookiePair = cookieArr[i].split('=');
      if (name === cookiePair[0].trim()) {
        return decodeURIComponent(cookiePair[1]); 
      }
    }
    return null; 
  }

  async function exec_service_rest(url, method_exec, data) {
    let answer; 
    if (method_exec === 'GET') {
      answer = await fetch(url)
    }else {
      answer = await fetch(url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-CSRFToken': getCookie('csrftoken')
        },
        body: JSON.stringify(data) // Convierte el objeto JS a String
      });
    }
    if (!answer.ok) {
        throw new Error(`Error en el servidor: ${answer.status} - ${answer.statusText}`);
    }    
    return await answer.json();
  }
  
  async function getCountries() {
    try {
      const data = await exec_service_rest('/contact/get_countries/', 'GET', {})
      const selectElement = document.getElementById('country'); 
      selectElement.innerHTML = ''; 
      for (let country of data.countries) {
        let newOption = document.createElement('option'); 
        newOption.value = country; 
        newOption.text = country; 
        selectElement.appendChild(newOption); 
      }
      getCities(); 
    } catch (error) {
      console.error(error); 
    } finally { }    
  }

  async function getCities() {    
    try {
      const selectCountry = document.getElementById('country'); 
      const selectCity = document.getElementById('city');

        {'country':selectCountry.value}); 
      selectCity.innerHTML = ''; 
      for (let city of data.cities) {
        let newOption = document.createElement('option'); 
        newOption.value = city; 
        newOption.text = city; 
        selectCity.appendChild(newOption); 
      }
    } catch (error) {
      console.error(error); 
    } finally { }    
  }

  document.addEventListener('DOMContentLoaded', function(){

  });
