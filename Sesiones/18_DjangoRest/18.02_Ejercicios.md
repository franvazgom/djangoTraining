1. data.py
```python
def get_projects(self, parameters=None):
        projects = Project.objects.all()
        if parameters:
            if 'title' in parameters:
                projects = projects.filter(title__contains = parameters['title'])
            if 'description' in parameters:
                projects = projects.filter(description__contains = parameters['description'])
        return projects
```
2. serializers.py
```python
class ProjectParameterSerializer(serializers.Serializer):

```
3. api_g.py
```python
from .serializers import SumSerializer, ProjectParameterSerializer
from django.core.serializers.json import DjangoJSONEncoder

class ProjectServices(APIView):
    def post(self, request):
        serializer = ProjectParameterSerializer(data = request.data)
        if serializer.is_valid():
            parameters = serializer.validated_data['parameters']
            data = Data()
            projects = data.get_projects(parameters)
            res = json.dumps(list(projects.values()), cls=DjangoJSONEncoder)
            return Response({'res':res}, status=status.HTTP_200_OK)
        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)
```
